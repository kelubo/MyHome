## 3. WiFi调试

### 3.1 WiFi扫描

```cpp
#include <WiFi.h>

void scanWiFi() {
    Serial.println("扫描WiFi网络...");
    int n = WiFi.scanNetworks();

    if (n == 0) {
        Serial.println("未找到网络");
    } else {
        Serial.print(n);
        Serial.println(" 个网络 found:");

        for (int i = 0; i < n; i++) {
            Serial.print(i + 1);
            Serial.print(": ");
            Serial.print(WiFi.SSID(i));
            Serial.print(" (");
            Serial.print(WiFi.RSSI(i));
            Serial.println(")");
        }
    }
}
```

### 3.2 WiFi连接测试

```cpp
void testWiFiConnection() {
    WiFi.mode(WIFI_STA);

    Serial.println("开始连接...");
    WiFi.begin("SSID", "PASSWORD");

    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 20) {
        delay(500);
        Serial.print(".");
        attempts++;
    }

    if (WiFi.status() == WL_CONNECTED) {
        Serial.println("\n连接成功!");
        Serial.print("IP: ");
        Serial.println(WiFi.localIP());
    } else {
        Serial.println("\n连接失败!");
    }
}
```

## 4. I2C调试

### 4.1 I2C扫描

```cpp
#include <Wire.h>

void scanI2C() {
    byte error, address;
    int nDevices;

    Serial.println("扫描I2C设备...");

    nDevices = 0;
    for (address = 1; address < 127; address++) {
        Wire.beginTransmission(address);
        error = Wire.endTransmission();

        if (error == 0) {
            Serial.print("找到设备 at 0x");
            if (address < 16) Serial.print("0");
            Serial.println(address, HEX);
            nDevices++;
        }
    }

    if (nDevices == 0) {
        Serial.println("未找到I2C设备");
    }
}
```

## 5. 传感器校准

### 5.1 MQ传感器校准

```cpp
void calibrateMQ() {
    Serial.println("开始校准...");
    Serial.println("请将传感器置于清洁空气中");

    delay(60000);  // 等待1分钟稳定

    float sum = 0;
    int samples = 100;

    for (int i = 0; i < samples; i++) {
        sum += analogRead(SENSOR_PIN);
        delay(100);
    }

    float avgRs = sum / samples;
    float r0 = avgRs * (5.0 / 1023.0) / 5.0;  // 假设在清洁空气中

    Serial.print("校准完成, R0 = ");
    Serial.println(r0);
}
```

## 8. 日志记录

```cpp
// 环形缓冲区日志
#define LOG_SIZE 50
char logBuffer[LOG_SIZE][100];
int logIndex = 0;
int logCount = 0;

void addLog(const char* message) {
    strncpy(logBuffer[logIndex], message, 99);
    logIndex = (logIndex + 1) % LOG_SIZE;
    if (logCount < LOG_SIZE) logCount++;
}

void printLog() {
    int start = (logCount < LOG_SIZE) ? 0 : logIndex;
    for (int i = 0; i < logCount; i++) {
        int idx = (start + i) % LOG_SIZE;
        Serial.println(logBuffer[idx]);
    }
}
```

## 9. 性能分析

```cpp
unsigned long loopStart;
unsigned long sensorTime;
unsigned long displayTime;
unsigned long wifiTime;

void loop() {
    loopStart = micros();

    // 传感器读取
    sensorTime = micros() - loopStart;
    displayTime = micros() - loopStart - sensorTime;
    wifiTime = micros() - loopStart - sensorTime - displayTime;

    // 串口输出性能数据
    Serial.print("Sensor: ");
    Serial.print(sensorTime);
    Serial.print("us, Display: ");
    Serial.print(displayTime);
    Serial.print("us, WiFi: ");
    Serial.print(wifiTime);
    Serial.println("us");
}
```

## 10. 常见问题排查表

| 现象 | 可能原因 | 解决方案 |
|------|----------|----------|
| 无输出 | 电源问题 | 检查5V电源 |
| 读数不变 | 传感器故障 | 更换传感器 |
| 数值跳变 | 干扰/接触不良 | 滤波/检查接线 |
| 编译失败 | 库缺失/版本 | 安装/更新库 |
| 上传失败 | 驱动/端口 | 重装驱动/换端口 |
